    function getTopicName(id){

    function renderTrainTopic(){

        qs.forEach((q,i)=>{

    function checkTrainTopic(){

        const qs = allTopics[currentTrainTopic];

        let score=0, newErr=[];

        for(let i=0;i<qs.length;i++){

            let sel = document.querySelector(`input[name="t${i}"]:checked`);

            let ans = sel?sel.value:"";

            currentTrainAnswers[i] = ans;

            if(ans===qs[i].correct) score++;

            else if(ans) newErr.push({topic:getTopicName(currentTrainTopic),question:qs[i].text,userAnswer:ans,correctAnswer:qs[i].correct,date:new Date().toLocaleString()});

        updateXP(score);

    }
